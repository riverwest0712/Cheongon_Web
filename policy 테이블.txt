-- -----------------------------------------------------
-- Table `cheongon`.`policy`
-- -----------------------------------------------------
DROP TABLE IF EXISTS `cheongon`.`policy` ;

CREATE TABLE IF NOT EXISTS `cheongon`.`policy` (
  `policy_id` VARCHAR(45) NOT NULL,
  `policy_name` VARCHAR(255) NULL,
  `category` VARCHAR(45) NULL,
  `category_main` VARCHAR(45) NULL,
  `category_sub` VARCHAR(45) NULL,
  `description` LONGTEXT NULL,
  `support_content` LONGTEXT NULL,
  `min_age` INT NULL,
  `max_age` INT NULL,
  `min_income` INT NULL,
  `max_income` INT NULL,
  `region` VARCHAR(45) NULL,
  `organization` VARCHAR(100) NULL,
  `apply_url` LONGTEXT NULL,
  `start_date` VARCHAR(45) NULL,
  `end_date` VARCHAR(45) NULL,
  `zip_code` VARCHAR(100) NULL,
  `created_at` DATETIME NULL,
  PRIMARY KEY (`policy_id`))
ENGINE = InnoDB;
